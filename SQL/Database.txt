1.Create a database using SQL query = CREATE DATABASE clg_db;

2.Use the database In SQL Management = USE clg_db;

3.Create a Table In Database = CREATE TABLE students (Stu_id int PRIMARY KEY NOT NULL,
						    Stu_name varchar(50),
						    Stu_age int(3));

4. Insert Data into the table = INSERT INTO students 
				Values  (1,"Mohit",20),
					(2,"Dhananjay",21),
					(3,"Divyansh",21);

5.Show data from the table = SELECT * FROM students;
